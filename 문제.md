    p.player_name 선수명,  
    COALESCE(p.height::text || ' cm', '0') 키,  
    COALESCE(p.weight::text || ' kg', '0') 몸무게  
FROM player p JOIN team t  
using (team_id)  
WHERE t.region_name = '수원'  
ORDER BY p.height DESC NULLS LAST;

SELECT t.team_name, s.stadium_name, (
        SELECT team_name
        FROM team t2
        WHERE t2.team_id = a.awayteam_id) 원정팀, a.sche_date
FROM schedule a
JOIN stadium s ON a.stadium_id = s.stadium_id
JOIN team t ON s.hometeam_id = t.team_id
JOIN player p ON t.team_id = p.team_id
WHERE a.sche_date = '2012-03-17'
GROUP BY t.team_name, s.stadium_name, a.awayteam_id, a.sche_date
ORDER BY t.team_name COLLATE "C" ASC;

SELECT 
    s.stadium_name,
    a.sche_date,
    t.team_name home_team,
    (
        SELECT team_name
        FROM team t2
        WHERE t2.team_id = a.awayteam_id
    ) away_team,home_score,away_score
FROM team t 
JOIN stadium s
  ON t.team_id = s.hometeam_id
JOIN schedule a
  ON t.team_id = a.hometeam_id
WHERE a.home_score - a.away_score >= 3;
